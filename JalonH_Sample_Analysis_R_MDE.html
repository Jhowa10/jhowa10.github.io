I. Summary

  • Report Name: Michigan Department of Education (MDE) Report

  • Frequency: Quarterly

  • Scope

    • Timeframe: Previous quarter and cumulative to-date

    • Population: Schools that launched (i.e had at least one trainee complete a core training type) under 31p2 contract

  • Output Metrics:

    • Number of schools implementing TRAILS in any Tier

    • Number of students who gained access to TRAILS resources and supports 

    • Number of staff trained by Tier (can be counted duplicatively and cumulatively)

      • Staff may be counted more than once when participating in more than one tier of programming

  • Report SOP

  • Data Sources

    • List of MDE Schools -> Salesforce Contract Implementations with Account info & Core Training Launch Date filtered where contract contains ‘31p2’ and by appropriate date parameters

    • Trainings by Tier -> Salesforce Training Affiliations filtered by List of MDE Schools

    • Student Access Data -> National Center for Education Statistics (NCES) search with the following criteria:

      • State: “Michigan”

      • School Description: “Regular, Special Education, Vocational, Other/Alternative”

    # Coalesce Implementing School and School Info
    school_or_implementing_name = coalesce(aff_school, aff_implementing_school),
    school_or_implementing_code = gsub("[^[:alnum:]]", "", coalesce(aff_school_code, aff_implementing_school_code),),
    # Training Date
    training_date = as.Date(training_string, format = "%m/%d/%Y"),
    # Determine MDE Role based on Training and T2 Role
    mde_role = case_when(
      # Tier 1 Roles
      training_tier == 'Tier 1' & training_topic == 'Curriculum' ~ 'T1_Staff',
      # Tier 2 Roles
      training_tier == 'Tier 2' & training_topic == 'CBT & Mindfulness' & current_tier_2_role %in% c('CIT', 'Coach') ~ 'T2_CIT',
      training_tier == 'Tier 2' & training_topic == 'CBT & Mindfulness' & current_tier_2_role %in% c('SMHP') ~ 'T2_SMHP',
      # Tier 3 Roles
      training_tier == 'Tier 3' & training_topic == 'ISD/Administrator' ~ 'T3_Admin',
      training_tier == 'Tier 3' & training_topic == 'School Staff' ~ 'T3_Educator', 
      training_tier == 'Tier 3' & training_topic == 'SMHP' ~ 'T3_SMHP', 
      training_tier == 'Tier 3' & training_topic == 'Community' ~ 'T3_Community', 
      training_tier == 'Tier 3' & training_topic == 'Risk Assessment and Management' ~ 'T3_Both_Admin_SMHP',
      .default = 'No MDE Role'),
    # Flag for different Implementing School
    flag_novel_imp_school = case_when(
      !is.na(aff_school) & !is.na(aff_implementing_school) & aff_school != aff_implementing_school ~ TRUE,
      .default = FALSE,
    )
  ) %>%
  filter(
    # Qualifying training
    training_topic %in% names(dict_mde_role),
    # Exclude records before 07/01/2023
    training_date >= as.Date("07/01/2023", format = "%m/%d/%Y")
    )

            right = as.Date(dict_report_period[[r_quarter]][['end']], format = "%Y-%m-%d")
            )
  )
